# The remote checker command, run on each login node by check_all_retry.sh.
# THIS FILE IS MEANT TO CHANGE. Claude Code / the autonomous agent should keep it
# current — it is the one place the checker evolves. Lines starting with # are
# stripped before sending.
#
# Keep current:
#   * the grep -vE exclusion list — add each newly BANKED solution's file id, or
#     "NEW FOUND?" cries wolf on old banked files every single run;
#   * the rung label + globs when the ladder climbs (n=32 -> n=33 ...);
#   * the GATE PROBES section — whatever probe is live this round.
#
# NOTE: do NOT use `exit` in here. The driver wraps this in BEGIN/END markers and
# an early exit would drop the END marker, making a good run look like a failure.

squeue -u dangord -h -o "%.14i %.10j %.2t %.11L %R" 2>/dev/null
cd $SCRATCH/bs45 2>/dev/null || echo "(cannot cd to \$SCRATCH/bs45)"
echo "--- NEW FOUND? ---"
grep -l "FOUND" sa_ladder_*.txt 2>/dev/null | grep -vE "46274622_4|14923090_[26]|16945067_3" || echo "(none yet)"
echo "--- n=32 progress ---"
for f in $(ls -t sa_ladder_*.txt 2>/dev/null | head -3); do hdr=$(grep -oE "BS\([0-9]+,[0-9]+\)" "$f" | head -1); best=$(grep -oE "bestAB=[0-9]+" "$f" | sort -t= -k2 -n | head -1 | grep -oE "[0-9]+$"); echo "$(basename $f) [$hdr] bestAB_min=$best | $(tail -1 "$f" | cut -c1-55)"; done
echo "--- GATE PROBES ---"
for f in $(ls -t wz_match_output_*.txt 2>/dev/null | head -2); do echo "=== $f ==="; grep -A5 "SUMMARY (n=" "$f" || tail -3 "$f"; done
echo "--- P22 GATE SHARDS (Nibi 17518826; sum valid ONLY at shards_reported=20) ---"
if ls pair22_gate_output_*.txt >/dev/null 2>&1; then grep -h SHARD_STREAM pair22_gate_output_*.txt 2>/dev/null | awk '{s+=$5; n++} END {print "shards_reported=" n+0 " partial_CD_stream_sum=" s+0}'; else echo "(no gate output files on this cluster)"; fi
